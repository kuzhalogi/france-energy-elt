CREATE TABLE dim_date (
    date date,
    date_key integer,
    year integer,
    quarter integer,
    month integer,
    day_of_month integer,
    day_of_week integer,
    day_name text,
    month_name text,
    is_weekend boolean
);

CREATE TABLE dim_region (
    region_code integer,
    region_name text
);

CREATE TABLE fact_consumption_daily (
    date date,
    region_code integer,
    consumption_mwh numeric,
    avg_load_mw numeric,
    peak_load_mw numeric,
    min_load_mw numeric,
    n_intervals bigint
);

CREATE TABLE fact_production_annual (
    year integer,
    region_code integer,
    nuclear_gwh numeric,
    thermal_gwh numeric,
    hydro_gwh numeric,
    wind_gwh numeric,
    solar_gwh numeric,
    bioenergy_gwh numeric,
    total_production_gwh numeric
);

CREATE TABLE fact_weather_daily (
    date date,
    region_code integer,
    t_min_c numeric,
    t_max_c numeric,
    t_mean_c numeric
);
